% --- Slide 2: The Problem ---
\begin{frame}
    \frametitle{The Interconnect Scaling Wall}

    \begin{alertblock}{Current Bottleneck: Laser Arrays Don't Scale}
        \textbf{Electrical interconnects are hitting their limits:}
        \begin{itemize}
            \item \textbf{Power:} $\sim$50\% of chip power is for data movement.
            \item \textbf{Bandwidth:} Limited by RC delays and crosstalk.
            \item \textbf{Heat:} I²R losses generate massive heat.
        \end{itemize}
    \end{alertblock}

    \begin{block}{Optical Solution Exists, BUT...}
        Current laser-array-based optical interconnects face scaling challenges.
        \begin{itemize}
            \item \textbf{Cost:} 64 wavelengths = 64 lasers = \$3,200--\$6,400.
            \item \textbf{Size:} $\sim$1000 mm².
            \item \textbf{Power:} 10-20W.
            \item \textbf{Scalability:} Impractical beyond 32 wavelengths.
        \end{itemize}
    \end{block}
\end{frame}

% --- Slide 4: What is a Frequency Comb ---
\begin{frame}
    \frametitle{What is a Frequency Comb?}
    
    \begin{block}{A light source with many equally-spaced optical frequencies}
        A frequency comb is like a ruler made of light, where each mark is a perfectly spaced, phase-coherent laser line.
    \end{block}
    
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{Traditional Combs}
                \begin{itemize}
                    \item Room-sized or benchtop
                    \item Expensive (\$50K--\$500K)
                    \item Power hungry
                \end{itemize}
            \end{alertblock}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{Soliton Microcombs}
                \begin{itemize}
                    \item Chip-scale (mm²)
                    \item Low cost (\$500--\$2K)
                    \item Low power (mW)
                    \item CMOS-compatible
                \end{itemize}
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 5: Why CW Pumping is Revolutionary ---
\begin{frame}
    \frametitle{Why CW Pumping is Revolutionary}
    
    \begin{block}{Key Advantage: Continuous Wave Input $\rightarrow$ Pulsed Output}
        You pump the microresonator with a simple, steady CW laser, and due to nonlinear physics, you automatically get a train of stable pulses out.
    \end{block}
    
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{Old Paradigm}
                To get pulses, you needed a complex, bulky, and expensive mode-locked laser.
            \end{alertblock}
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{block}{New Paradigm}
                A simple CW laser plus a micro-chip creates the pulses automatically. This is the key to making combs compact, cheap, and power-efficient.
            \end{block}
        \end{column}
    \end{columns}
\end{frame}

% --- Slide 10: Technical Challenges \& Solutions ---
\begin{frame}
    \frametitle{Technical Challenges \& Our Solutions}
    
    \begin{columns}[T]
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{Challenge 1: Soliton Stabilization}
                Solitons are metastable and sensitive to thermal drift.
            \end{alertblock}
            \textbf{Solution:} Automated stabilization using feedback control with a single PID feedback loop and auto-locking algorithm.
        \end{column}
        \begin{column}{0.5\textwidth}
            \begin{alertblock}{Challenge 2: Pump Laser Integration}
                The pump laser is a different material (III-V) than the Si chip.
            \end{alertblock}
            \textbf{Solution:} Hybrid approach. External laser for Gen 1, heterogeneous integration (bonding) for Gen 2.
        \end{column}
    \end{columns}
    
    \begin{alertblock}{Challenge 3: Manufacturing Variability}
        Dispersion engineering requires precision; not all chips will work perfectly.
    \end{alertblock}
    \textbf{Solution:} Robust designs with wide operating ranges, post-fabrication tuning, and partnership with proven foundries.
\end{frame}
